## Laboratorio · la escalera de seis peldaños

```javascript
function arnes(cal) {
  const previas = (cal.preexistentes || []).map((p) => ({ clave: p.clave, payload: JSON.stringify(p.payload), id: p.id }));
  const ops = previas.slice();
  const guion = (cal.eventos || []).slice();
  const consultas = (cal.consultas || []).slice();
  const max = cal.maxIntentos || 3;
  let llamadas = 0, conflictoServido = false, corrida = 0, roto = false;
  const porCorrida = [], timeoutsPorCorrida = [], desconocidosPorCorrida = [];
  function aplicar(clave, payload) {
    const p = JSON.stringify(payload);
    const previa = clave == null ? null : ops.find((o) => o.clave === clave);
    if (previa) return previa.payload === p ? { tipo: "ok", id: previa.id } : { tipo: "conflicto" };
    const id = "op-" + (ops.length - previas.length + 1);
    ops.push({ clave, payload: p, id });
    return { tipo: "ok", id };
  }
  const servicio = {
    enviar(clave, payload) {
      llamadas++; porCorrida[corrida]++;
      if (llamadas > 20) throw new Error("bucle de reintentos");
      const ev = guion.length ? guion.shift() : "ok";
      if (ev === "transitorio") return { tipo: "error", permanente: false };
      if (ev === "permanente") return { tipo: "error", permanente: true };
      const r = aplicar(clave == null ? null : String(clave), payload);
      if (r.tipo === "conflicto") { conflictoServido = true; return r; }
      if (ev === "aplica-y-pierde" || ev === "tarde") { timeoutsPorCorrida[corrida]++; return { tipo: "timeout" }; }
      return r;
    },
    consultar(clave) {
      const ev = consultas.length ? consultas.shift() : "auto";
      if (ev === "timeout") return { tipo: "timeout" };
      const o = ops.find((x) => x.clave === String(clave));
      return o ? { tipo: "encontrada", id: o.id } : { tipo: "no-encontrada" };
    },
    registrar(e) { if (e && e.estado === "desconocido") desconocidosPorCorrida[corrida]++; },
  };
  const resultados = [];
  const corridas = cal.corridas || 1;
  for (corrida = 0; corrida < corridas; corrida++) {
    porCorrida[corrida] = 0; timeoutsPorCorrida[corrida] = 0; desconocidosPorCorrida[corrida] = 0;
    try {
      resultados.push(enviarOrden(servicio, JSON.parse(JSON.stringify(cal.orden)), { maxIntentos: max }) || {});
    } catch (e) {
      roto = true; resultados.push({});
    }
  }
  const contadas = cal.descartarPrimera ? resultados.slice(1) : resultados;
  const nuevas = ops.slice(previas.length);
  const reconcilia = (r) => conflictoServido
    ? r.estado === "conflicto"
    : nuevas.length > 0 ? r.estado === "confirmada" && nuevas.some((o) => o.id === r.id) : r.estado === "rechazada";
  return {
    una_operacion: ops.length === 1,
    conflicto_visible: conflictoServido ? contadas.every((r) => r.estado === "conflicto") : contadas.every((r) => r.estado !== "conflicto"),
    reintentos_acotados: !roto && porCorrida.every((n) => n <= max),
    desconocido_registrado: timeoutsPorCorrida.every((t, i) => t === 0 || desconocidosPorCorrida[i] >= t),
    reconciliado: contadas.every(reconcilia),
  };
}
```

### Solución
```javascript
function enviarOrden(servicio, orden, politica) {
  const clave = orden.clave;
  let huboTimeout = false;
  for (let intento = 1; intento <= politica.maxIntentos; intento++) {
    const r = servicio.enviar(clave, orden.payload);
    if (r.tipo === "ok") {
      servicio.registrar({ intento, resultado: "ok", estado: "confirmada" });
      return { estado: "confirmada", id: r.id };
    }
    if (r.tipo === "conflicto") {
      servicio.registrar({ intento, resultado: "conflicto", estado: "conflicto" });
      return { estado: "conflicto" };
    }
    if (r.tipo === "error" && r.permanente) {
      servicio.registrar({ intento, resultado: "error", estado: "rechazada" });
      return { estado: "rechazada" };
    }
    if (r.tipo === "timeout") {
      huboTimeout = true;
      servicio.registrar({ intento, resultado: "timeout", estado: "desconocido" });
    } else {
      servicio.registrar({ intento, resultado: "error", estado: "no-recibida" });
    }
  }
  if (!huboTimeout) return { estado: "rechazada" };
  const c = servicio.consultar(clave);
  if (c.tipo === "encontrada") return { estado: "confirmada", id: c.id };
  if (c.tipo === "no-encontrada") return { estado: "rechazada" };
  servicio.registrar({ intento: 0, resultado: "consulta", estado: "desconocido" });
  return { estado: "desconocida" };
}
```

## ¿Cero, una o dos órdenes?
type: prediccion
id: s202-pred-timeout
rol: prediccion

Tu cliente manda una orden sin clave de idempotencia. El servicio no responde a tiempo y el cliente la vuelve a mandar igual; esta vez llega `ok`. Antes de ver la respuesta, predice cuántas órdenes puede haber registrado el servicio.

### Opciones
- [x] Una o dos, y desde el cliente no puedes distinguirlas: el primer envío pudo aplicarse antes de que se perdiera su respuesta.
- Exactamente una: el `ok` del segundo envío confirma que el primero falló.
- Exactamente dos: todo timeout implica que el servicio aplicó la petición.
- Cero o una: el segundo envío reemplaza al primero si los dos tienen el mismo contenido.

### Justificación
Un timeout sólo dice que el cliente dejó de esperar. Si el primer envío murió antes de llegar, hay una orden; si se aplicó y la respuesta se perdió, hay dos, porque sin clave el servicio no tiene cómo reconocer la repetición. El `ok` del segundo envío habla del segundo, no del primero. Que el contenido sea igual no basta: para el servicio, dos peticiones sin identificador común son dos órdenes. Es el motivo por el que la clave existe.

## Mil clientes vuelven a la vez
type: prediccion
id: s202-pred-jitter
rol: prediccion

Un servicio se cae 3 segundos. Mil clientes fallan en el mismo instante y todos reintentan con retardo exponencial exacto: 1 s, 2 s, 4 s. Predice qué ve el servicio cuando vuelve.

### Opciones
- [x] Picos sincronizados: los mil reintentos llegan juntos en cada escalón; sumar fluctuación aleatoria a la espera los repartiría.
- Una carga pareja, porque el retardo exponencial ya reparte los reintentos.
- Nada distinto de lo normal: tres reintentos por cliente no alcanzan para sobrecargar.
- Ningún reintento: el retardo exponencial cancela los intentos después del primero.

### Justificación
Si todos fallaron a la vez y todos esperan exactamente lo mismo, todos vuelven a la vez. El retardo exponencial espacia los intentos de *cada* cliente, pero no separa a los clientes entre sí. El artículo de timeouts y reintentos de la ficha presenta la fluctuación (*jitter*) justo para eso: una componente aleatoria en la espera reparte los reintentos y evita repetir el pico.

## Tres capas que reintentan
type: prediccion
id: s202-pred-capas
rol: prediccion

Una petición del usuario atraviesa tres servicios en cadena: la interfaz llama a A, A llama a B y B llama a la base de datos. Cada capa hace hasta 3 intentos ante un error. La base de datos está caída. Predice cuántas llamadas puede recibir la base de datos por una sola petición del usuario.

### Opciones
- [x] Hasta 27: los intentos se multiplican capa por capa (3 × 3 × 3).
- Hasta 9: sólo cuentan las dos capas más cercanas a la base de datos.
- Hasta 3: cada capa reintenta lo suyo y la base de datos ve sólo los de B.
- Hasta 1: la primera falla corta la cadena entera.

### Justificación
Cada intento de la interfaz dispara hasta 3 intentos de A, y cada uno de esos dispara hasta 3 de B contra la base de datos: 3 × 3 × 3 = 27. Es la amplificación que describe el artículo de reintentos de la ficha, y la razón para reintentar en un solo punto de la pila y no en todas las capas.

## La misma clave, otro contenido
type: prediccion
id: s202-pred-payload
rol: prediccion

Un servicio implementa claves de idempotencia. Llega una petición con la clave `ord-77`, que ya existe para una orden de 1 unidad, pero ahora pide 5. Predice qué debería responder un servicio bien diseñado.

### Opciones
- [x] Un error explícito de conflicto entre la clave y el contenido, sin aplicar nada.
- La respuesta guardada de la primera orden (1 unidad), porque la clave ya existe.
- Una orden nueva de 5 unidades, porque el contenido es distinto.
- Actualizar la orden existente a 5 unidades.

### Justificación
Devolver en silencio el resultado viejo haría creer al cliente que su orden de 5 quedó hecha. Crear otra orden rompe el contrato de la clave. Actualizar la existente cambia una operación ya confirmada sin que nadie lo pidiera. El artículo de reintentos seguros de la ficha trata este caso como un error que el servicio debe devolver. En la escalera del banco, es el peldaño 4.

## Cuántos intentos caben en el presupuesto
type: code
id: s202-code-presupuesto
rol: practica

El número de reintentos sale del presupuesto de latencia, no de una constante. Calcula cuántos intentos caben.

### Especificación
`intentosQueCaben(presupuestoMs, timeoutMs, baseMs, topeMs)`:
- Cada intento puede tardar hasta `timeoutMs`.
- Antes del intento `i` (para `i ≥ 2`) se espera `min(topeMs, baseMs × 2^(i−2))`: `baseMs`, luego el doble, con tope.
- Devuelve el mayor `n ≥ 0` tal que la suma de `n` intentos y sus esperas no pase de `presupuestoMs`.
- Sin fluctuación: el cálculo es el peor caso determinista.

### Firma
```javascript

  // devuelve un entero >= 0
}
```

### Solución
```javascript
function intentosQueCaben(presupuestoMs, timeoutMs, baseMs, topeMs) {
  let total = 0;
  let n = 0;
  for (let i = 1; ; i++) {
    const espera = i === 1 ? 0 : Math.min(topeMs, baseMs * Math.pow(2, i - 2));
    if (total + espera + timeoutMs > presupuestoMs) return n;
    total += espera + timeoutMs;
    n = i;
    if (n > 10000) return n;
  }
}
```

### Pistas
- Suma intento por intento y detente antes de pasarte, no después.
- Con presupuesto menor que un solo timeout, no cabe ningún intento.

## Extensión · tres responsables del envío
type: production
id: s202-ext-responsables
rol: practica

Compara tres formas de mandar la orden del laboratorio: una llamada síncrona desde la interfaz, un worker que consume una cola, y un outbox local leído por un proceso aparte. Para cada una, explica qué componente se hace responsable del envío, en qué ventana puede perderse o duplicarse la orden, y dónde viven la clave de idempotencia y el reconciliador.

### Modelo
Llamada síncrona: responde el que llama, a menudo una persona con un botón. La ventana peligrosa es el timeout: la duda queda en la interfaz, que tiene que mostrar «desconocida» y no ofrecer «reintentar» como orden nueva. La clave nace con la orden en el cliente y el reconciliador es un proceso del servidor que revisa las desconocidas. Worker con cola: responde el worker. La cola entrega al menos una vez, así que dos workers pueden tomar el mismo mensaje y uno puede caerse después de enviar; la clave viaja en el mensaje y el servicio deduplica. El reconciliador revisa mensajes que salieron sin confirmación. Outbox local: responde el proceso que lee el outbox. Como la orden y su mensaje se escriben en la misma transacción local, no se pierde nada entre decidir y avisar, pero el envío puede repetirse tras una caída; la clave sale del outbox y el reconciliador recorre las entradas enviadas sin confirmar. En los tres, la garantía de una sola operación la da el servicio con la clave; el transporte no es exactly-once.

### Regla
La idempotencia por clave hace inocuas las repeticiones, no las evita. Cambiar de transporte cambia quién responde por el envío y dónde mirar cuando algo queda desconocido; nunca elimina la clave ni el reconciliador.

### Rúbrica
- Nombra para cada opción el componente responsable del envío.
- Identifica una ventana de pérdida o duplicado concreta en cada una.
- Ubica la clave de idempotencia y el reconciliador en cada diseño.
- Declara que ningún transporte vuelve exactly-once la operación de extremo a extremo.

## Giro · el proveedor sin claves
type: giro
id: s202-giro-sin-clave
rol: transferencia

La dependencia cambia. El proveedor migra a una API heredada que **no acepta claves de idempotencia** ni consulta por clave. Lo único que ofrece es un listado de las órdenes del día, cada una con un campo libre de «referencia del cliente» de hasta 32 caracteres. Tu consumidor ya está en producción y los timeouts siguen ocurriendo. ¿Qué cambias para seguir teniendo una sola operación lógica por orden, qué garantía pierdes y qué le muestras al usuario mientras no sabes?

### Rúbrica
- No reintenta a ciegas tras un timeout: sin clave en el servicio, reintentar sin comprobar duplica.
- Usa la referencia del cliente para llevar su propia clave en el payload y reconcilia contra el listado antes de reenviar.
- Declara la garantía que se pierde (la deduplicación ya no es atómica en el servicio; queda una ventana entre listar y reenviar) y cómo la acota.
- Mantiene «desconocida» visible para el usuario hasta reconciliar y nombra qué haría si el listado tampoco responde.
